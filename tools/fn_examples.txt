#   glob: <SAS code>   placed before the DATA step (or PROC SQL)

today() = date() => as.numeric(sas_f_today() == sas_f_date())
today() > '01JAN2024'd => as.numeric(sas_f_today() > 23376)
weekday(today()) <= 7 => as.numeric(sas_f_weekday(sas_f_today()) <= 7)

## SQLCASE
mode: sql
case when 2 > 1 then 'yes' else 'no' end => sas_f_sqlcase(2 > 1, "yes", "no")
case when 1 > 2 then 1 when 2 > 1 then 2 else 3 end => sas_f_sqlcase(1 > 2, 1, 2 > 1, 2, 3)
case when 1 > 2 then 1 else 3 end => sas_f_sqlcase(1 > 2, 1, 3)

## SQLINLIST
mode: sql
2 in (1, 2, 3) => sas_f_sqlinlist(2, 1, 2, 3)
'b' in ('a', 'c') => sas_f_sqlinlist("b", "a", "c")
. in (., 1) => sas_f_sqlinlist(NA, NA, 1)

## SQLLIKE
mode: sql
'abc' like 'a%' => as.numeric(sas_f_sqllike("abc", "a%"))
'abc' like '_b_' => as.numeric(sas_f_sqllike("abc", "_b_"))
'abc' like 'b%' => as.numeric(sas_f_sqllike("abc", "b%"))
'a%c' like 'a\%c' escape '\' => as.numeric(sas_f_sqllike("a%c", "a\\%c", "\\"))

## SQLQUANT
mode: sql
glob: data work.t; v = 2; output; v = 3; output; run;
1 < any (select v from work.t) => sas_f_sqlquant(1, c(2, 3), "<", "any")
3 > all (select v from work.t) => sas_f_sqlquant(3, c(2, 3), ">", "all")
3 >= all (select v from work.t) => sas_f_sqlquant(3, c(2, 3), ">=", "all")
