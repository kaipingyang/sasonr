# ---- DATA CHG (SAS line 12) -- level L2
# L2: BY-group processing (FIRST./LAST.); retained value carried forward (baseline / LOCF)
.df <- sas_rt_ds(work, "lab") |>
  sas_rt_note_read("lab") |>
  sas_rt_check_sorted("usubjid", FALSE) |>
  dplyr::mutate(
    first.usubjid = sas_rt_first(usubjid),
    last.usubjid = sas_rt_last(usubjid),
    .cv_base_1 = aval,
    .cs_base_1 = sas_rt_true(first.usubjid),
    base = sas_rt_carry(list(.cv_base_1), list(.cs_base_1), NA_real_),
    chg = sas_rt_arith(aval, base, "-", .at = c(17L, 14L))
  )
work$chg <- .df |> dplyr::select(usubjid, visit, aval, base, chg)
